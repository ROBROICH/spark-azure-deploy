1. Get the application name:
  {{- if .Values.ingress.enabled }}
  {{- range $host := .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ range $host.paths }}{{ .path }}{{ end }}
  {{- end }}
  {{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ include "temporal-worker.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "temporal-worker.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ include "temporal-worker.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
  {{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch its status by running 'kubectl get --namespace {{ include "temporal-worker.namespace" . }} svc -w {{ include "temporal-worker.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ include "temporal-worker.namespace" . }} {{ include "temporal-worker.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
  {{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ include "temporal-worker.namespace" . }} -l "app.kubernetes.io/name={{ include "temporal-worker.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ include "temporal-worker.namespace" . }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ include "temporal-worker.namespace" . }} port-forward $POD_NAME 8080:$CONTAINER_PORT
  {{- end }}
